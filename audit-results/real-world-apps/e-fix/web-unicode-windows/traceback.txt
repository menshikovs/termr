{
    target <- Sys.getenv("TERMR_E_RUNTIME", "")
    if (nzchar(target)) 
        writeLines(capture.output(sys.calls()), file.path(dirname(target), 
            "traceback.txt"))
    .efix_record_runtime("error")
    quit(save = "no", status = 1L, runLast = FALSE)
})()
